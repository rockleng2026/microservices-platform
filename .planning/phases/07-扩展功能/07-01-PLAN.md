```java
// Add new field:
private final RedissonClient redisson;

// Add script constant at top of class:
private static final String LUA_PREALLOCATE_SCRIPT =
    "local stock = redis.call('GET', KEYS[1])\n" +
    "if stock == false then\n" +
    "    return -1\n" +
    "end\n" +
    "local current = tonumber(stock)\n" +
    "local quantity = tonumber(ARGV[1])\n" +
    "if current < quantity then\n" +
    "    return -1\n" +
    "end\n" +
    "local remaining = current - quantity\n" +
    "redis.call('DECRBY', KEYS[1], quantity)\n" +
    "return remaining\n";

// Replace decrement-check-rollback block with:
String stockKey = SKU_STOCK_KEY_PREFIX + skuId;
org.redisson.client.RedisResponse<Integer> resp = redisson.getScript()
    .evalReadOnly(
        org.redisson.client.script.RedisScriptSource.fromString(LUA_PREALLOCATE_SCRIPT),
        org.redisson.client.script.RScriptReturnType.INTEGER,
        java.util.List.of(stockKey),
        quantity
    );
Integer result = resp.get(); // blocks until Lua returns
if (result != null && result < 0) {
    log.warn("Stock insufficient for SKU {}: requested {}", skuId, quantity);
    return false;
}
Long stockAfterDecr = result.longValue(); // remaining stock after allocation
```

Note: The Lua script performs DECRBY and returns remaining stock directly. The stockAfterDecr variable still captures the post-decrement value for the stock log.
