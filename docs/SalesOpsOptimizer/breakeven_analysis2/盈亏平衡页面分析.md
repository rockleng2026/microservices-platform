const calculateSeriesValue = (series, xValue, context) => {
  switch (series.seriesType) {
    case 'fixed':
      return Number(series.seriesValue) || 0;
      
    case 'variable':
      return context[series.seriesValue] || 0;
      
    case 'formula':
      // 构建包含X轴值的计算上下文
      const formulaContext = {
        ...context,
        x: { value: xValue, type: 'NUMBER' }
      };
      return calculatorEngine.evaluate(series.seriesValue, formulaContext);
  }
};
