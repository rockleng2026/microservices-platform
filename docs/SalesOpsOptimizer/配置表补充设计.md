# 盈亏平衡分析模块 - 配置表补充设计

## 1. 系统配置表设计

### 1.1 系统配置表 (soo_system_config)

```sql
CREATE TABLE `soo_system_config` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
  `config_key` VARCHAR(100) NOT NULL COMMENT '配置键',
  `config_value` TEXT NOT NULL COMMENT '配置值',
  `config_type` VARCHAR(20) NOT NULL DEFAULT 'string' COMMENT '配置类型',
  `config_group` VARCHAR(50) DEFAULT 'default' COMMENT '配置分组',
  `config_category` VARCHAR(50) DEFAULT 'general' COMMENT '配置类别',
  `tenant_id` BIGINT DEFAULT 0 COMMENT '租户ID(0表示全局配置)',
  
  -- 配置描述和约束
  `description` VARCHAR(500) COMMENT '配置描述',
  `default_value` TEXT COMMENT '默认值',
  `value_constraints` JSON COMMENT '值约束规则',
  `validation_regex` VARCHAR(500) COMMENT '验证正则表达式',
  
  -- 配置状态
  `is_active` TINYINT(1) DEFAULT 1 COMMENT '是否启用',
  `is_editable` TINYINT(1) DEFAULT 1 COMMENT '是否可编辑',
  `is_sensitive` TINYINT(1) DEFAULT 0 COMMENT '是否敏感配置',
  `requires_restart` TINYINT(1) DEFAULT 0 COMMENT '是否需要重启生效',
  
  -- 环境和版本
  `environment` VARCHAR(20) DEFAULT 'all' COMMENT '适用环境',
  `min_version` VARCHAR(20) COMMENT '最小版本要求',
  `max_version` VARCHAR(20) COMMENT '最大版本要求',
  
  -- 操作记录
  `last_modified_by` BIGINT COMMENT '最后修改人',
  `last_modified_name` VARCHAR(100) COMMENT '最后修改人姓名',
  `modification_reason` VARCHAR(200) COMMENT '修改原因',
  
  -- 审计字段
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_config_key_tenant` (`config_key`, `tenant_id`),
  KEY `idx_config_group` (`config_group`),
  KEY `idx_config_category` (`config_category`),
  KEY `idx_tenant_active` (`tenant_id`, `is_active`),
  KEY `idx_environment` (`environment`),
  KEY `idx_updated_at` (`updated_at`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='系统配置表';
```

## 2. 场景模板表设计

### 2.1 场景模板表 (soo_scenario_templates)

```sql
CREATE TABLE `soo_scenario_templates` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
  `template_id` VARCHAR(50) NOT NULL COMMENT '模板ID',
  `template_name` VARCHAR(200) NOT NULL COMMENT '模板名称',
  `template_type` VARCHAR(50) NOT NULL COMMENT '模板类型',
  `template_category` VARCHAR(50) DEFAULT 'default' COMMENT '模板分类',
  `tenant_id` BIGINT DEFAULT 0 COMMENT '租户ID(0表示全局模板)',
  
  -- 模板内容
  `template_config` JSON NOT NULL COMMENT '模板配置',
  `parameter_defaults` JSON COMMENT '参数默认值',
  `calculation_rules` JSON COMMENT '计算规则',
  `validation_rules` JSON COMMENT '验证规则',
  
  -- 模板描述
  `description` TEXT COMMENT '模板描述',
  `usage_scenario` TEXT COMMENT '使用场景说明',
  `expected_outcomes` TEXT COMMENT '预期结果说明',
  `best_practices` TEXT COMMENT '最佳实践建议',
  
  -- 适用性配置
  `industry_types` JSON COMMENT '适用行业类型',
  `company_sizes` JSON COMMENT '适用公司规模',
  `business_models` JSON COMMENT '适用业务模式',
  `complexity_level` VARCHAR(20) DEFAULT 'medium' COMMENT '复杂度等级',
  
  -- 模板状态
  `is_active` TINYINT(1) DEFAULT 1 COMMENT '是否启用',
  `is_public` TINYINT(1) DEFAULT 1 COMMENT '是否公开可用',
  `is_editable` TINYINT(1) DEFAULT 1 COMMENT '是否可编辑',
  `is_deletable` TINYINT(1) DEFAULT 1 COMMENT '是否可删除',
  
  -- 使用统计
  `usage_count` INT DEFAULT 0 COMMENT '使用次数',
  `success_rate` DECIMAL(5,4) DEFAULT 0.0000 COMMENT '成功率',
  `avg_satisfaction` DECIMAL(3,2) DEFAULT 0.00 COMMENT '平均满意度',
  `last_used_at` TIMESTAMP COMMENT '最后使用时间',
  
  -- 版本控制
  `version` VARCHAR(20) DEFAULT '1.0.0' COMMENT '模板版本',
  `parent_template_id` VARCHAR(50) COMMENT '父模板ID',
  `version_notes` TEXT COMMENT '版本说明',
  
  -- 创建者信息
  `creator_id` BIGINT COMMENT '创建者ID',
  `creator_name` VARCHAR(100) COMMENT '创建者姓名',
  `creator_type` VARCHAR(20) DEFAULT 'user' COMMENT '创建者类型',
  
  -- 审计字段
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_template_id` (`template_id`),
  KEY `idx_template_type` (`template_type`),
  KEY `idx_template_category` (`template_category`),
  KEY `idx_tenant_public` (`tenant_id`, `is_public`, `is_active`),
  KEY `idx_complexity` (`complexity_level`),
  KEY `idx_usage_stats` (`usage_count` DESC, `success_rate` DESC),
  KEY `idx_created_at` (`created_at`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='场景模板表';
```

## 3. 验证规则表设计

### 3.1 验证规则表 (soo_validation_rules)

```sql
CREATE TABLE `soo_validation_rules` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
  `rule_id` VARCHAR(50) NOT NULL COMMENT '规则ID',
  `rule_name` VARCHAR(200) NOT NULL COMMENT '规则名称',
  `rule_type` VARCHAR(50) NOT NULL COMMENT '规则类型',
  `rule_category` VARCHAR(50) DEFAULT 'parameter' COMMENT '规则分类',
  `tenant_id` BIGINT DEFAULT 0 COMMENT '租户ID(0表示全局规则)',
  
  -- 规则配置
  `rule_config` JSON NOT NULL COMMENT '规则配置',
  `validation_expression` TEXT COMMENT '验证表达式',
  `error_message` VARCHAR(500) NOT NULL COMMENT '错误提示信息',
  `warning_message` VARCHAR(500) COMMENT '警告提示信息',
  `suggestion_message` VARCHAR(500) COMMENT '建议提示信息',
  
  -- 适用范围
  `applicable_parameters` JSON COMMENT '适用参数列表',
  `applicable_scenarios` JSON COMMENT '适用场景列表',
  `exclusion_conditions` JSON COMMENT '排除条件',
  
  -- 规则属性
  `severity_level` VARCHAR(20) DEFAULT 'error' COMMENT '严重级别',
  `validation_timing` VARCHAR(50) DEFAULT 'on_change' COMMENT '验证时机',
  `is_blocking` TINYINT(1) DEFAULT 1 COMMENT '是否阻塞操作',
  `auto_fix_enabled` TINYINT(1) DEFAULT 0 COMMENT '是否启用自动修复',
  `auto_fix_config` JSON COMMENT '自动修复配置',
  
  -- 业务规则
  `business_context` TEXT COMMENT '业务上下文说明',
  `industry_standards` JSON COMMENT '行业标准参考',
  `regulatory_requirements` JSON COMMENT '法规要求',
  
  -- 规则状态
  `is_active` TINYINT(1) DEFAULT 1 COMMENT '是否启用',
  `is_customizable` TINYINT(1) DEFAULT 1 COMMENT '是否可自定义',
  `priority` INT DEFAULT 5 COMMENT '优先级(1-10)',
  
  -- 执行统计
  `execution_count` INT DEFAULT 0 COMMENT '执行次数',
  `violation_count` INT DEFAULT 0 COMMENT '违规次数',
  `last_executed_at` TIMESTAMP COMMENT '最后执行时间',
  `avg_execution_time` INT COMMENT '平均执行时间(毫秒)',
  
  -- 版本信息
  `version` VARCHAR(20) DEFAULT '1.0.0' COMMENT '规则版本',
  `version_notes` TEXT COMMENT '版本说明',
  
  -- 创建者信息
  `creator_id` BIGINT COMMENT '创建者ID',
  `creator_name` VARCHAR(100) COMMENT '创建者姓名',
  
  -- 审计字段
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_rule_id` (`rule_id`),
  KEY `idx_rule_type` (`rule_type`),
  KEY `idx_rule_category` (`rule_category`),
  KEY `idx_tenant_active` (`tenant_id`, `is_active`),
  KEY `idx_severity_priority` (`severity_level`, `priority`),
  KEY `idx_violation_stats` (`violation_count` DESC, `execution_count` DESC)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='验证规则表';
```

### 5.1 获取配置值的函数

```sql
DELIMITER //

CREATE FUNCTION `get_config_value`(
  p_config_key VARCHAR(100),
  p_tenant_id BIGINT DEFAULT 0,
  p_default_value TEXT DEFAULT NULL
) RETURNS TEXT
READS SQL DATA
DETERMINISTIC
COMMENT '获取配置值，支持租户级别和默认值'
BEGIN
  DECLARE v_config_value TEXT DEFAULT NULL;
  
  -- 首先查找租户级配置
  IF p_tenant_id > 0 THEN
    SELECT config_value INTO v_config_value
    FROM soo_system_config
    WHERE config_key = p_config_key 
      AND tenant_id = p_tenant_id 
      AND is_active = 1
    LIMIT 1;
  END IF;
  
  -- 如果没有找到，查找全局配置
  IF v_config_value IS NULL THEN
    SELECT config_value INTO v_config_value
    FROM soo_system_config
    WHERE config_key = p_config_key 
      AND tenant_id = 0 
      AND is_active = 1
    LIMIT 1;
  END IF;
  
  -- 如果还是没有找到，返回默认值
  IF v_config_value IS NULL THEN
    SET v_config_value = p_default_value;
  END IF;
  
  RETURN v_config_value;
END //

DELIMITER ;
```

### 5.2 配置更新存储过程

```sql
DELIMITER //

CREATE PROCEDURE `update_config_with_history`(
  IN p_config_key VARCHAR(100),
  IN p_new_value TEXT,
  IN p_tenant_id BIGINT DEFAULT 0,
  IN p_operator_id BIGINT,
  IN p_operator_name VARCHAR(100),
  IN p_change_reason VARCHAR(500),
  OUT p_success BOOLEAN,
  OUT p_message VARCHAR(500)
)
BEGIN
  DECLARE v_old_value TEXT DEFAULT NULL;
  DECLARE v_change_id VARCHAR(50);
  DECLARE v_exit_flag BOOLEAN DEFAULT FALSE;
  
  -- 错误处理
  DECLARE CONTINUE HANDLER FOR SQLEXCEPTION
  BEGIN
    SET v_exit_flag = TRUE;
    SET p_success = FALSE;
    SET p_message = '配置更新失败';
    ROLLBACK;
  END;
  
  START TRANSACTION;
  
  -- 生成变更ID
  SET v_change_id = CONCAT('CFG_', DATE_FORMAT(NOW(), '%Y%m%d%H%i%s'), '_', 
                          LPAD(CONNECTION_ID(), 6, '0'));
  
  -- 获取当前值
  SELECT config_value INTO v_old_value
  FROM soo_system_config
  WHERE config_key = p_config_key AND tenant_id = p_tenant_id;
  
  -- 更新配置
  IF v_old_value IS NOT NULL THEN
    UPDATE soo_system_config
    SET config_value = p_new_value,
        last_modified_by = p_operator_id,
        last_modified_name = p_operator_name,
        modification_reason = p_change_reason
    WHERE config_key = p_config_key AND tenant_id = p_tenant_id;
  ELSE
    -- 配置不存在时插入新记录
    INSERT INTO soo_system_config (
      config_key, config_value, tenant_id,
      last_modified_by, last_modified_name, modification_reason
    ) VALUES (
      p_config_key, p_new_value, p_tenant_id,
      p_operator_id, p_operator_name, p_change_reason
    );
  END IF;
  
  -- 记录变更历史
  INSERT INTO soo_config_change_history (
    change_id, config_type, config_key, tenant_id,
    old_value, new_value, change_type, change_reason,
    operator_id, operator_name
  ) VALUES (
    v_change_id, 'system_config', p_config_key, p_tenant_id,
    v_old_value, p_new_value, 
    CASE WHEN v_old_value IS NULL THEN 'create' ELSE 'update' END,
    p_change_reason, p_operator_id, p_operator_name
  );
  
  IF v_exit_flag THEN
    SET p_success = FALSE;
  ELSE
    SET p_success = TRUE;
    SET p_message = '配置更新成功';
    COMMIT;
  END IF;
  
END //

DELIMITER ;
```

## 6. 配置表视图

### 6.1 活跃配置视图

```sql
CREATE VIEW `v_active_configs` AS
SELECT 
  c.config_key,
  c.config_value,
  c.config_type,
  c.config_group,
  c.config_category,
  c.tenant_id,
  c.description,
  c.is_editable,
  c.is_sensitive,
  c.updated_at,
  
  -- 使用统计
  COALESCE(s.access_count, 0) as today_access_count,
  COALESCE(s.unique_users, 0) as today_unique_users,
  
  -- 变更信息
  h.last_change_time,
  h.last_operator_name

FROM soo_system_config c

LEFT JOIN soo_config_usage_stats s ON (
  c.config_key = s.config_key 
  AND c.tenant_id = s.tenant_id 
  AND s.stat_date = CURDATE()
)

LEFT JOIN (
  SELECT 
    config_key,
    tenant_id,
    MAX(created_at) as last_change_time,
    (SELECT operator_name FROM soo_config_change_history h2 
     WHERE h2.config_key = h1.config_key 
       AND h2.tenant_id = h1.tenant_id
     ORDER BY created_at DESC LIMIT 1) as last_operator_name
  FROM soo_config_change_history h1
  GROUP BY config_key, tenant_id
) h ON (c.config_key = h.config_key AND c.tenant_id = h.tenant_id)

WHERE c.is_active = 1
ORDER BY c.config_group, c.config_category, c.config_key;
```
