// Run in node to add flowcharts based on rules
const data = require('./api-docs.json');

function getTemplate(endpoint) {
  const { method, path, businessLogic } = endpoint;
  // Refund flow
  if (path.includes('refund') || businessLogic.includes('退款')) return 'refund';
  // Auth flow
  if (path.includes('auth') || businessLogic.includes('登录') || businessLogic.includes('注册')) return 'auth';
  // Delete
  if (method === 'DELETE') return 'delete';
  // Update
  if (method === 'PUT') return 'update';
  // Create
  if (method === 'POST') return 'action';
  // List query
  if (endpoint.parameters.some(p => p.name === 'page') || businessLogic.includes('列表')) return 'list';
  // Simple query
  return 'query';
}
