# 盈策通决策平台 - 盈亏平衡试算与预测功能详细设计

## 1. 核心设计理念

### 1.1 全联动计算模型
基于销售运营优化器的盈亏平衡计算表设计，构建一个"牵一发而动全身"的动态计算系统：

- **任何参数变化都会触发全局重算**
- **多层级联动验证机制**
- **智能合理性调整系数**
- **实时预测模型重新生成**

### 1.2 设计原则
1. **数据驱动**: 所有决策基于实时计算结果
2. **场景模拟**: 支持多种假设条件下的快速试算
3. **风险预警**: 自动识别不合理的参数组合
4. **智能建议**: 基于计算结果提供优化建议

```javascript
// 盈亏平衡点计算核心算法
class BreakevenCalculationEngine {
    
    /**
     * 计算盈亏平衡点
     * @param {Object} params - 计算参数
     * @returns {Object} 计算结果
     */
    calculateBreakevenPoint(params) {
        const {
            period,
            grossMargin,
            payrollData,
            operatingCosts,
            teamData
        } = params;

        // 1. 计算总固定成本 (基于工资计算结果表)
        const totalFixedCost = this.calculateTotalFixedCost(payrollData);
        
        // 2. 计算变动成本率
        const variableCostRatio = this.calculateVariableCostRatio(payrollData, params.currentRevenue);
        
        // 3. 计算盈亏平衡点
        const breakevenPoint = totalFixedCost / (grossMargin - variableCostRatio);
        
        return {
            breakevenPoint,
            totalFixedCost,
            variableCostRatio,
            marginSafety: params.currentRevenue - breakevenPoint,
            marginSafetyRatio: (params.currentRevenue - breakevenPoint) / params.currentRevenue
        };
    }

    /**
     * 计算总固定成本
     */
    calculateTotalFixedCost(payrollData) {
        return {
            // 固定运营成本 + 基础工资及绩效总额
            fixedOperatingCost: payrollData.fixedOperatingCost,
            totalBaseSalary: payrollData.employees.reduce((sum, emp) => 
                sum + emp.adjustedBaseSalary, 0),
            totalPerformancePay: payrollData.employees.reduce((sum, emp) => 
                sum + emp.performancePay, 0),
            totalFixed: payrollData.fixedOperatingCost + 
                       payrollData.employees.reduce((sum, emp) => 
                           sum + emp.adjustedBaseSalary + emp.performancePay, 0)
        };
    }

    /**
     * 计算变动成本率
     */
    calculateVariableCostRatio(payrollData, currentRevenue) {
        if (currentRevenue === 0) return 0;
        
        const totalCommission = payrollData.employees.reduce((sum, emp) => 
            sum + emp.personalCommission + emp.teamCommission, 0);
        const totalBonus = payrollData.employees.reduce((sum, emp) => 
            sum + emp.departmentBonus, 0);
        
        return (totalCommission + totalBonus + payrollData.variableOperatingCost) / currentRevenue;
    }

    /**
     * 多场景盈亏平衡点计算
     */
    calculateMultiScenario(baseParams) {
        const scenarios = [
            { name: '保守场景', grossMargin: 0.10 },
            { name: '基准场景', grossMargin: 0.30 },
            { name: '乐观场景', grossMargin: 0.65 }
        ];

        return scenarios.map(scenario => {
            const result = this.calculateBreakevenPoint({
                ...baseParams,
                grossMargin: scenario.grossMargin
            });
            
            return {
                ...scenario,
                ...result,
                feasibilityScore: this.assessFeasibility(result, scenario.grossMargin)
            };
        });
    }

    /**
     * 可行性评估
     */
    assessFeasibility(result, grossMargin) {
        // 基于历史数据和行业标准评估可行性
        if (grossMargin >= 0.5) return 0.9;
        if (grossMargin >= 0.3) return 0.7;
        if (grossMargin >= 0.15) return 0.4;
        return 0.2;
    }
}
```

```javascript
// 联动计算管理器
class CascadeCalculationManager {
    
    constructor() {
        this.calculationChain = new Map();
        this.reasonabilityAdjuster = new ReasonabilityAdjuster();
        this.setupCalculationChain();
    }

    /**
     * 设置计算链
     */
    setupCalculationChain() {
        // 定义参数依赖关系
        this.calculationChain.set('baseSalary', [
            'adjustedBaseSalary', 'performancePay', 'socialSecurity', 'totalFixedCost'
        ]);
        this.calculationChain.set('performanceScore', [
            'performancePay', 'personalCommission', 'teamCommission'
        ]);
        this.calculationChain.set('projectRevenue', [
            'personalCommission', 'teamCommission', 'variableCostRatio'
        ]);
        this.calculationChain.set('teamSize', [
            'teamCommission', 'departmentBonus', 'perCapitaMetrics'
        ]);
    }

    /**
     * 参数变化处理
     */
    async handleParameterChange(paramName, newValue, context) {
        console.log(`参数变化: ${paramName} = ${newValue}`);
        
        // 1. 合理性检查
        const adjustedValue = await this.reasonabilityAdjuster.adjust(paramName, newValue, context);
        
        // 2. 触发联动计算
        const affectedParams = this.calculationChain.get(paramName) || [];
        const recalculationResults = {};
        
        for (const affectedParam of affectedParams) {
            recalculationResults[affectedParam] = await this.recalculate(affectedParam, context);
        }
        
        // 3. 重新生成预测
        const newForecast = await this.regenerateForecast(context);
        
        // 4. 发送更新事件
        this.emit('calculationComplete', {
            changed: paramName,
            adjustedValue,
            affected: recalculationResults,
            forecast: newForecast
        });
        
        return {
            success: true,
            adjustedValue,
            cascadeResults: recalculationResults,
            forecast: newForecast
        };
    }
}
```

```javascript
// 合理性调整器
class ReasonabilityAdjuster {
    
    constructor() {
        this.constraints = {
            baseSalary: { min: 3000, max: 100000 },
            performanceScore: { min: 0, max: 100 },
            grossMargin: { min: 0.05, max: 0.8 },
            teamSize: { min: 1, max: 50 }
        };
        
        this.industryBenchmarks = {
            techCompany: {
                avgGrossMargin: 0.65,
                avgSalaryCostRatio: 0.4,
                avgPerformanceRatio: 0.8
            }
        };
    }

    /**
     * 智能调整参数
     */
    async adjust(paramName, value, context) {
        const constraint = this.constraints[paramName];
        if (!constraint) return value;

        // 1. 基础约束检查
        let adjustedValue = Math.max(constraint.min, Math.min(constraint.max, value));
        
        // 2. 业务逻辑约束
        adjustedValue = await this.applyBusinessConstraints(paramName, adjustedValue, context);
        
        // 3. 行业基准对比
        const benchmarkAdjustment = await this.applyBenchmarkAdjustment(paramName, adjustedValue, context);
        
        // 4. 记录调整日志
        if (adjustedValue !== value) {
            console.warn(`参数 ${paramName} 从 ${value} 调整为 ${adjustedValue}`);
        }
        
        return benchmarkAdjustment;
    }

    /**
     * 应用业务逻辑约束
     */
    async applyBusinessConstraints(paramName, value, context) {
        switch (paramName) {
            case 'performanceScore':
                // 绩效分数不能导致绩效工资超过基础工资的120%
                const maxAllowedScore = this.calculateMaxPerformanceScore(context);
                return Math.min(value, maxAllowedScore);
                
            case 'teamSize':
                // 团队规模不能导致人均提成低于合理水平
                const minViableTeamSize = this.calculateMinViableTeamSize(context);
                return Math.max(value, minViableTeamSize);
                
            case 'grossMargin':
                // 毛利率变化不能导致盈亏平衡点过高
                return this.constrainGrossMarginByFeasibility(value, context);
                
            default:
                return value;
        }
    }

    /**
     * 计算最大允许绩效分数
     */
    calculateMaxPerformanceScore(context) {
        const { baseSalary, performanceRatioMax } = context;
        return performanceRatioMax * 100; // 转换为百分制
    }

    /**
     * 基于可行性约束毛利率
     */
    constrainGrossMarginByFeasibility(grossMargin, context) {
        const engine = new BreakevenCalculationEngine();
        const testResult = engine.calculateBreakevenPoint({
            ...context,
            grossMargin
        });
        
        // 如果盈亏平衡点过高（超过历史最高营业额的150%），降低毛利率要求
        const historicalMaxRevenue = context.historicalMaxRevenue || 1000000;
        if (testResult.breakevenPoint > historicalMaxRevenue * 1.5) {
            return this.findFeasibleGrossMargin(context, historicalMaxRevenue * 1.5);
        }
        
        return grossMargin;
    }
}
```

```javascript
class DynamicForecastEngine {
    
    constructor() {
        this.models = {
            exponentialSmoothing: new ExponentialSmoothingModel(),

        // 1. 影响评估
        const impactAssessment = this.assessParameterImpact(changedParams, context);

        const selectedModel = this.selectOptimalModel(impactAssessment, context);

        await selectedModel.retrain(context.historicalData, changedParams);
        
        // 4. 生成多场景预测
        const scenarios = await this.generateMultiScenarioForecast(selectedModel, context);
        
        // 5. 置信度评估
        const confidenceMetrics = this.calculateConfidenceMetrics(scenarios, context);
        
        return {
            model: selectedModel.name,
            scenarios,
            confidence: confidenceMetrics,
            recommendedActions: this.generateRecommendations(scenarios, context)
        };
    }

    /**
     * 生成多场景预测
     */
    async generateMultiScenarioForecast(model, context) {
        const scenarios = [
            { name: '悲观', factor: 0.8, probability: 0.2 },
            { name: '基准', factor: 1.0, probability: 0.6 },
            { name: '乐观', factor: 1.2, probability: 0.2 }
        ];

        const forecasts = {};
        
        for (const scenario of scenarios) {
            const adjustedContext = this.adjustContextForScenario(context, scenario);
            
            forecasts[scenario.name] = {
                revenue: await model.predictRevenue(adjustedContext),
                costs: await model.predictCosts(adjustedContext),
                breakeven: await model.predictBreakevenPoint(adjustedContext),
                probability: scenario.probability,
                keyDrivers: await model.identifyKeyDrivers(adjustedContext)
            };
        }
        
        return forecasts;
    }
}
```

```javascript
// 季节性分析模块
class SeasonalityAnalyzer {
    
    /**
     * 分析季节性模式
     */
    analyzeSeasonality(historicalData) {
        const monthlyPatterns = this.extractMonthlyPatterns(historicalData);
        const seasonalFactors = this.calculateSeasonalFactors(monthlyPatterns);
        const trendComponent = this.extractTrend(historicalData);
        
        return {
            seasonalFactors,
            trendComponent,
            cyclicalPatterns: this.identifyCyclicalPatterns(historicalData),
            peakSeasons: this.identifyPeakSeasons(seasonalFactors),
            recommendations: this.generateSeasonalRecommendations(seasonalFactors)
        };
    }

    /**
     * 提取月度模式
     */
    extractMonthlyPatterns(data) {
        const monthlyData = new Array(12).fill(0).map(() => []);
        
        data.forEach(record => {
            const month = new Date(record.date).getMonth();
            monthlyData[month].push(record.revenue);
        });
        
        return monthlyData.map((monthData, index) => ({
            month: index + 1,
            average: monthData.reduce((sum, val) => sum + val, 0) / monthData.length,
            variance: this.calculateVariance(monthData),
            trend: this.calculateMonthlyTrend(monthData)
        }));
    }

    /**
     * 生成季节性调整建议
     */
    generateSeasonalRecommendations(seasonalFactors) {
        const recommendations = [];
        
        seasonalFactors.forEach((factor, month) => {
            if (factor.factor > 1.2) {
                recommendations.push({
                    month: month + 1,
                    type: 'peak_season',
                    recommendation: `${month + 1}月为业绩高峰期，建议增加人员配置和营销投入`,
                    expectedImpact: '营业额提升15-25%'
                });
            } else if (factor.factor < 0.8) {
                recommendations.push({
                    month: month + 1,
                    type: 'low_season',
                    recommendation: `${month + 1}月为业绩低谷期，建议控制成本并开展培训`,
                    expectedImpact: '成本节约10-20%'
                });
            }
        });
        
        return recommendations;
    }
}
```

```jsx
// 智能参数控制组件
const ParameterControlPanel = ({ parameters, onParameterChange, validationRules }) => {
    const [localParams, setLocalParams] = useState(parameters);
    const [validationErrors, setValidationErrors] = useState({});
    const [adjustmentHistory, setAdjustmentHistory] = useState([]);

    const handleInputChange = async (paramName, value) => {
        setIsCalculating(true);
        
        try {
            // 实时验证
            const validation = await validateParameter(paramName, value, localParams);
            
            if (validation.hasWarnings) {
                // 显示警告但允许输入
                showWarningToast(validation.warnings);
            }
            
            if (validation.hasErrors) {
                setValidationErrors(prev => ({
                    ...prev,
                    [paramName]: validation.errors
                }));
                return;
            }
            
            // 清除错误
            setValidationErrors(prev => {
                const newErrors = { ...prev };
                delete newErrors[paramName];
                return newErrors;
            });
            
            // 更新本地状态
            const newParams = { ...localParams, [paramName]: value };
            setLocalParams(newParams);
            
            // 触发联动计算
            const result = await onParameterChange(paramName, value, newParams);
            
            // 记录调整历史
```

```jsx
                    <ChartTypeSelector
                        value={chartType}
                        onChange={setChartType}
                        options={Object.keys(chartConfig)}
                    />
                </div>
            </CardHeader>
            
            <CardContent>
                {chartType === 'breakeven_curve' && (
                    <BreakevenCurveChart
                        data={data}
                        onInteraction={handleChartInteraction}
                        selectedPoint={selectedDataPoint}
                    />
                )}
                
                {chartType === 'sensitivity_analysis' && (
                    <SensitivityAnalysisChart
                        data={data.sensitivityData}
                        onInteraction={handleChartInteraction}
                    />
                )}
                
                {chartType === 'scenario_comparison' && (
                    <ScenarioRadarChart
                        scenarios={data.scenarios}
                        metrics={['feasibility', 'profitability', 'sustainability']}
                    />
                )}
                
                {/* 图表说明和操作提示 */}
                <ChartLegend 
                    type={chartType}
                    interactive={chartConfig[chartType].interactive}
                />
            </CardContent>
        </Card>
    );
};
```

### 6.1 计算性能优化

```javascript
// 计算性能优化管理器
class CalculationPerformanceManager {
    
    constructor() {
        this.calculationQueue = new PriorityQueue();
        this.resultCache = new LRUCache(1000);
        this.debounceTimers = new Map();
    }

    /**
     * 防抖动计算
     */
    debouncedCalculation(key, calculationFn, delay = 300) {
        return new Promise((resolve, reject) => {
            // 清除之前的定时器
            if (this.debounceTimers.has(key)) {
                clearTimeout(this.debounceTimers.get(key));
            }
            
            // 设置新的定时器
            const timer = setTimeout(async () => {
                try {
                    const result = await calculationFn();
                    this.debounceTimers.delete(key);
                    resolve(result);
                } catch (error) {
                    this.debounceTimers.delete(key);
                    reject(error);
                }
            }, delay);
            
            this.debounceTimers.set(key, timer);
        });
    }

    /**
     * 增量计算优化
     */
    incrementalCalculation(previousResult, changedParams, context) {
        // 只重新计算受影响的部分
        const affectedComponents = this.identifyAffectedComponents(changedParams);
        
        let updatedResult = { ...previousResult };
        
        for (const component of affectedComponents) {
            updatedResult[component] = this.recalculateComponent(
                component, 
                changedParams, 
                context
            );
        }
        
        return updatedResult;
    }

    /**
     * 并行计算优化
     */
    async parallelCalculation(calculations) {
        const chunks = this.chunkCalculations(calculations);
        const results = [];
        
        for (const chunk of chunks) {
            const chunkResults = await Promise.all(
                chunk.map(calc => this.executeCalculation(calc))
            );
            results.push(...chunkResults);
        }
        
        return results;
    }
}
```
