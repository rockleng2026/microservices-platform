```java
package com.central.mall.service.impl;

import com.baomidou.mybatisplus.extension.service.impl.ServiceImpl;
import com.central.mall.mapper.MallFaqMapper;
import com.central.mall.model.entity.MallFaq;
import com.central.mall.service.IFaqService;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import java.util.*;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
public class FaqServiceImpl extends ServiceImpl<MallFaqMapper, MallFaq> implements IFaqService {

    private static final Map<String, String> CATEGORY_LABELS = Map.of(
        "dev", "开发类",
        "biz", "业务类",
        "config", "配置类",
        "error", "常见错误"
    );

    @Override
    public List<Map<String, String>> getCategories() {
        return CATEGORY_LABELS.entrySet().stream()
            .map(e -> Map.of("key", e.getKey(), "label", e.getValue()))
            .collect(Collectors.toList());
    }

    @Override
    public List<MallFaq> getFaqsByCategory(String category) {
        return lambdaQuery()
            .eq(MallFaq::getCategory, category)
            .eq(MallFaq::getStatus, 1)
            .orderByAsc(MallFaq::getSortOrder)
            .list();
    }

    @Override
    public List<MallFaq> getAllFaqs() {
        return lambdaQuery()
            .eq(MallFaq::getStatus, 1)
            .orderByAsc(MallFaq::getCategory)
            .orderByAsc(MallFaq::getSortOrder)
            .list();
    }
}
```

```java
package com.central.mall.controller;

import com.central.common.model.Result;
import com.central.mall.model.entity.MallFaq;
import com.central.mall.service.IFaqService;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import lombok.RequiredArgsConstructor;
import org.springframework.web.bind.annotation.*;

import java.util.List;
import java.util.Map;

@RestController
@RequestMapping("/api/mall/faqs")
@RequiredArgsConstructor
@Tag(name = "FAQ管理", description = "商城FAQ接口")
public class FaqController {

    private final IFaqService faqService;

    @GetMapping("/categories")
    @Operation(summary = "获取FAQ分类列表")
    public Result<List<Map<String, String>>> getCategories() {
        return Result.succeed(faqService.getCategories());
    }

    @GetMapping
    @Operation(summary = "获取FAQ列表")
    public Result<List<MallFaq>> getFaqs(
            @RequestParam(required = false) String category) {
        if (category != null && !category.isEmpty()) {
            return Result.succeed(faqService.getFaqsByCategory(category));
        }
        return Result.succeed(faqService.getAllFaqs());
    }

    @GetMapping("/{id}")
    @Operation(summary = "获取单条FAQ详情")
    public Result<MallFaq> getFaqById(@PathVariable Long id) {
        MallFaq faq = faqService.getById(id);
        if (faq == null) {
            return Result.failed("FAQ不存在");
        }
        return Result.succeed(faq);
    }
}
```

```typescript
/**
 * FAQ API Service - HELP-03
 * Endpoints for FAQ page in portal
 */
import { request } from '@/utils/request';

// FAQ DTO
export interface FaqDTO {
  id?: number;
  category: string;
  question: string;
  cause?: string;
  solution?: string;
  sortOrder?: number;
  status: number;
  viewCount?: number;
}

// Category DTO
export interface CategoryDTO {
  key: string;
  label: string;
}

// API response with data wrapper
export interface ApiResponse<T> {
  code: number;
  msg?: string;
  datas?: T;
}

/**
 * Get FAQ categories
 */
export async function getFaqCategories(): Promise<CategoryDTO[]> {
  const response = await request<ApiResponse<CategoryDTO[]>>('/api-mall/api/mall/faqs/categories', {
    method: 'GET',
  });
  return response.datas || [];
}

/**
 * Get FAQ list (optionally filtered by category)
 */
export async function getFaqList(category?: string): Promise<FaqDTO[]> {
  const params = category ? { category } : {};
  const response = await request<ApiResponse<FaqDTO[]>>('/api-mall/api/mall/faqs', {
    method: 'GET',
    params,
  });
  return response.datas || [];
}

/**
 * Get single FAQ by ID
 */
export async function getFaqById(id: number): Promise<FaqDTO | null> {
  const response = await request<ApiResponse<FaqDTO>>(`/api-mall/api/mall/faqs/${id}`, {
    method: 'GET',
  });
  return response.datas || null;
}
```
