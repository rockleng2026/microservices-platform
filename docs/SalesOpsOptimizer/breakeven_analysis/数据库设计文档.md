# 盈亏平衡分析模块 - 数据库设计文档

## 1. 数据库概览

### 1.1 设计原则
- **数据一致性**: 保证联动计算的数据完整性
- **高性能**: 支持实时计算和频繁查询
- **可扩展性**: 支持多租户和大数据量
- **版本控制**: 记录参数变更历史
- **缓存友好**: 优化查询性能

### 1.2 技术选型
- **主数据库**: MySQL 8.0
- **缓存数据库**: Redis 6.0
- **时间序列数据**: InfluxDB (可选)
- **搜索引擎**: Elasticsearch (可选)

## 4. 存储过程和函数

### 4.1 盈亏平衡点计算函数

```sql
DELIMITER //

CREATE FUNCTION `calc_breakeven_point`(
  p_fixed_cost DECIMAL(15,2),
  p_gross_margin DECIMAL(8,6),
  p_variable_cost_ratio DECIMAL(8,6)
) RETURNS DECIMAL(15,2)
READS SQL DATA
DETERMINISTIC
COMMENT '计算盈亏平衡点'
BEGIN
  DECLARE v_result DECIMAL(15,2);
  DECLARE v_denominator DECIMAL(8,6);
  
  -- 检查参数有效性
  IF p_gross_margin <= 0 OR p_variable_cost_ratio < 0 THEN
    RETURN NULL;
  END IF;
  
  SET v_denominator = p_gross_margin - p_variable_cost_ratio;
  
  -- 检查分母是否为正数
  IF v_denominator <= 0 THEN
    RETURN NULL;
  END IF;
  
  SET v_result = p_fixed_cost / v_denominator;
  
  RETURN v_result;
END //

DELIMITER ;
```

### 4.2 参数变更批量处理存储过程

```sql
DELIMITER //

CREATE PROCEDURE `batch_update_parameters`(
  IN p_analysis_id VARCHAR(50),
  IN p_parameter_changes JSON,
  IN p_user_id BIGINT,
  IN p_user_name VARCHAR(100),
  IN p_change_reason VARCHAR(200),
  OUT p_change_batch_id VARCHAR(50),
  OUT p_calculation_result JSON
)
BEGIN
  DECLARE v_tenant_id BIGINT;
  DECLARE v_old_parameters JSON;
  DECLARE v_new_parameters JSON;
  DECLARE v_exit_flag BOOLEAN DEFAULT FALSE;
  DECLARE v_error_msg VARCHAR(500);
  
  -- 错误处理
  DECLARE CONTINUE HANDLER FOR SQLEXCEPTION
  BEGIN
    SET v_exit_flag = TRUE;
    GET DIAGNOSTICS CONDITION 1
      v_error_msg = MESSAGE_TEXT;
    ROLLBACK;
  END;
  
  START TRANSACTION;
  
  -- 生成变更批次ID
  SET p_change_batch_id = CONCAT('CHG_', DATE_FORMAT(NOW(), '%Y%m%d%H%i%s'), '_', 
                                 LPAD(CONNECTION_ID(), 6, '0'));
  
  -- 获取当前分析信息
  SELECT tenant_id, current_parameters 
  INTO v_tenant_id, v_old_parameters
  FROM soo_breakeven_analysis 
  WHERE analysis_id = p_analysis_id;
  
  IF v_tenant_id IS NULL THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = '分析不存在';
  END IF;
  
  -- 应用参数变更
  SET v_new_parameters = JSON_MERGE_PATCH(v_old_parameters, p_parameter_changes);
  
  -- 记录参数变更日志
  INSERT INTO soo_parameter_change_log (
    analysis_id, change_batch_id, tenant_id,
    parameter_name, old_value, new_value,
    change_source, change_reason, user_id, user_name
  )
  SELECT 
    p_analysis_id, p_change_batch_id, v_tenant_id,
    param_key, 
    JSON_UNQUOTE(JSON_EXTRACT(v_old_parameters, CONCAT('$.', param_key))),
    JSON_UNQUOTE(JSON_EXTRACT(v_new_parameters, CONCAT('$.', param_key))),
    'manual', p_change_reason, p_user_id, p_user_name
  FROM JSON_TABLE(p_parameter_changes, '$.*' COLUMNS (
    param_key VARCHAR(100) PATH '$'
  )) AS jt;
  
  -- 更新分析参数
  UPDATE soo_breakeven_analysis 
  SET 
    current_parameters = v_new_parameters,
    last_recalculation = NOW(),
    version = version + 1
  WHERE analysis_id = p_analysis_id;
  
  -- 触发重新计算
  CALL `trigger_recalculation`(p_analysis_id, p_change_batch_id, p_calculation_result);
  
  IF v_exit_flag THEN
    SET p_change_batch_id = NULL;
    SET p_calculation_result = JSON_OBJECT('error', v_error_msg);
  ELSE
    COMMIT;
  END IF;
  
END //

DELIMITER ;
```

## 6. 视图定义

### 6.1 分析概览视图

```sql
CREATE VIEW `v_analysis_overview` AS
SELECT 
  a.analysis_id,
  a.analysis_name,
  a.analysis_type,
  a.analysis_period,
  a.tenant_id,
  a.creator_name,
  a.breakeven_point,
  a.margin_safety,
  a.margin_safety_ratio,
  a.status,
  a.created_at,
  a.last_recalculation,
  
  -- 场景统计
  s.scenario_count,
  s.recommended_scenario,
  s.best_feasibility_score,
  s.worst_risk_level,
  
  -- 参数变更统计
  p.recent_changes_count,
  p.last_change_time,
  p.most_changed_parameter,
  
  -- 计算性能
  c.avg_calculation_time,
  c.last_calculation_status,
  
  -- 风险评估
  CASE 
    WHEN a.margin_safety_ratio > 0.3 THEN '低风险'
    WHEN a.margin_safety_ratio > 0.15 THEN '中风险'
    ELSE '高风险'
  END AS risk_assessment

FROM soo_breakeven_analysis a

LEFT JOIN (
  SELECT 
    analysis_id,
    COUNT(*) as scenario_count,
    MAX(CASE WHEN is_recommended = 1 THEN scenario_name END) as recommended_scenario,
    MAX(feasibility_score) as best_feasibility_score,
    MAX(CASE WHEN risk_level = '高' THEN 1 WHEN risk_level = '中' THEN 2 ELSE 3 END) as worst_risk_level
  FROM soo_breakeven_scenarios
  WHERE status = 'active'
  GROUP BY analysis_id
) s ON a.analysis_id = s.analysis_id

LEFT JOIN (
  SELECT 
    analysis_id,
    COUNT(*) as recent_changes_count,
    MAX(created_at) as last_change_time,
    (SELECT parameter_name FROM soo_parameter_change_log p2 
     WHERE p2.analysis_id = p1.analysis_id 
     GROUP BY parameter_name 
     ORDER BY COUNT(*) DESC LIMIT 1) as most_changed_parameter
  FROM soo_parameter_change_log p1
  WHERE created_at >= DATE_SUB(NOW(), INTERVAL 7 DAY)
  GROUP BY analysis_id
) p ON a.analysis_id = p.analysis_id

LEFT JOIN (
  SELECT 
    analysis_id,
    AVG(calculation_time) as avg_calculation_time,
    (SELECT status FROM soo_calculation_status c2 
     WHERE c2.analysis_id = c1.analysis_id 
     ORDER BY created_at DESC LIMIT 1) as last_calculation_status
  FROM soo_calculation_status c1
  WHERE created_at >= DATE_SUB(NOW(), INTERVAL 1 DAY)
  GROUP BY analysis_id
) c ON a.analysis_id = c.analysis_id;
```

### 6.2 性能监控视图

```sql
CREATE VIEW `v_performance_metrics` AS
SELECT 
  DATE(created_at) as metric_date,
  task_type,
  COUNT(*) as total_calculations,
  COUNT(CASE WHEN status = 'completed' THEN 1 END) as successful_calculations,
  COUNT(CASE WHEN status = 'failed' THEN 1 END) as failed_calculations,
  AVG(calculation_time) as avg_calculation_time,
  MIN(calculation_time) as min_calculation_time,
  MAX(calculation_time) as max_calculation_time,
  PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY calculation_time) as p95_calculation_time,
  AVG(memory_usage) as avg_memory_usage,
  MAX(memory_usage) as max_memory_usage,
  SUM(CASE WHEN cache_hit = 1 THEN 1 ELSE 0 END) / COUNT(*) * 100 as cache_hit_rate

FROM soo_calculation_status
WHERE created_at >= DATE_SUB(NOW(), INTERVAL 30 DAY)
GROUP BY DATE(created_at), task_type
ORDER BY metric_date DESC, task_type;
```

DELIMITER //

CREATE PROCEDURE `migrate_legacy_data`()
BEGIN
  DECLARE done INT DEFAULT FALSE;
  DECLARE v_old_id BIGINT;
  DECLARE v_old_data JSON;

    SELECT id, analysis_data FROM legacy_breakeven_table;

  START TRANSACTION;

  read_loop: LOOP

      analysis_name,
      current_parameters,

      JSON_EXTRACT(v_old_data, '$.parameters'),
      JSON_UNQUOTE(JSON_EXTRACT(v_old_data, '$.result.breakevenPoint')),
      NOW()
    );
    
  END LOOP;
  
  CLOSE cur;
  COMMIT;
  
END //

DELIMITER //

CREATE PROCEDURE `check_data_consistency`()
BEGIN
  DECLARE v_inconsistency_count INT DEFAULT 0;
  
  -- 检查分析表和日志表的一致性
  SELECT COUNT(*) INTO v_inconsistency_count
  FROM soo_parameter_change_log l
  LEFT JOIN soo_breakeven_analysis a ON l.analysis_id = a.analysis_id
  WHERE a.analysis_id IS NULL;
  
  IF v_inconsistency_count > 0 THEN
    INSERT INTO system_alerts (alert_type, alert_message, severity, created_at)
    VALUES ('data_consistency', CONCAT('发现 ', v_inconsistency_count, ' 条孤立的参数变更记录'), 'warning', NOW());
  END IF;
  
  -- 检查场景表的一致性
  SELECT COUNT(*) INTO v_inconsistency_count
  FROM soo_breakeven_scenarios s
  LEFT JOIN soo_breakeven_analysis a ON s.analysis_id = a.analysis_id
  WHERE a.analysis_id IS NULL;
  
  IF v_inconsistency_count > 0 THEN
    INSERT INTO system_alerts (alert_type, alert_message, severity, created_at)
    VALUES ('data_consistency', CONCAT('发现 ', v_inconsistency_count, ' 条孤立的场景记录'), 'warning', NOW());
  END IF;
  
END //

DELIMITER ;
