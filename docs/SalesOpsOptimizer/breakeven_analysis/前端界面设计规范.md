# 盈亏平衡分析前端界面设计规范

## 1. 设计系统概览

### 1.1 设计原则
- **数据可视化优先**: 复杂的财务数据通过直观的图表展示
- **实时响应**: 参数调整后立即显示计算结果
- **渐进式展示**: 从概览到详细，支持钻取分析
- **智能交互**: 提供参数建议和合理性检查
- **移动友好**: 响应式设计，支持多设备访问

### 1.2 技术栈
- **前端框架**: React 18 + TypeScript
- **UI组件库**: Ant Design + Custom Components
- **图表库**: ECharts + D3.js
- **状态管理**: Zustand
- **样式方案**: Tailwind CSS + CSS Modules
- **构建工具**: Vite

### 2.2 智能数字输入组件 (SmartNumberInput)

```tsx
interface SmartNumberInputProps {
  label: string;
  value: number;
  onChange: (value: number) => void;
  formatter?: 'currency' | 'percentage' | 'number';
  rules?: ValidationRule[];
  helpText?: string;
  suffix?: string;
  realTimeValidation?: boolean;
}

const SmartNumberInput: React.FC<SmartNumberInputProps> = ({
  label,
  value,
  onChange,
  formatter = 'number',
  rules = [],
  helpText,
  suffix,
  realTimeValidation = true
}) => {
  const [localValue, setLocalValue] = useState(value);
  const [validationState, setValidationState] = useState<ValidationState>('normal');
  const [errorMessage, setErrorMessage] = useState('');
  const [suggestions, setSuggestions] = useState<string[]>([]);
  
  const debouncedOnChange = useDebouncedCallback(onChange, 500);
  
  const handleValueChange = async (newValue: number) => {
    setLocalValue(newValue);
    
    if (realTimeValidation) {
      const validation = await validateValue(newValue, rules);
      setValidationState(validation.state);
      setErrorMessage(validation.message);
      setSuggestions(validation.suggestions);
    }
    
    debouncedOnChange(newValue);
  };
  
  const formatValue = (val: number): string => {
    switch (formatter) {
      case 'currency':
        return new Intl.NumberFormat('zh-CN', {
          style: 'currency',
          currency: 'CNY'
        }).format(val);
      case 'percentage':
        return `${(val * 100).toFixed(2)}%`;
      default:
        return val.toLocaleString();
    }
  };
  
  return (
    <Form.Item
      label={
        <span className="flex items-center gap-2">
          {label}
          {helpText && (
            <Tooltip title={helpText}>
              <InfoCircleOutlined className="text-gray-400" />
            </Tooltip>
          )}
        </span>
      }
      validateStatus={validationState}
      help={errorMessage}
      extra={suggestions.length > 0 && (
        <div className="mt-2">
          <Text type="secondary" className="text-sm">建议:</Text>
          <ul className="text-sm text-gray-600 mt-1">
            {suggestions.map((suggestion, index) => (
              <li key={index}>• {suggestion}</li>
            ))}
          </ul>
        </div>
      )}
    >
      <InputNumber
        value={localValue}
        onChange={handleValueChange}
        formatter={formatValue}
        parser={(value) => Number(value?.replace(/[^\d.-]/g, '') || 0)}
        suffix={suffix}
        className="w-full"
        precision={formatter === 'percentage' ? 4 : 2}
      />
    </Form.Item>
  );
};
```

### 2.3 实时结果展示卡片 (MetricCard)

```tsx
interface MetricCardProps {
  title: string;
  value: number;
  formatter?: 'currency' | 'percentage' | 'number';
  trend?: {
    direction: 'up' | 'down' | 'stable';
    value: number;
    period: string;
  };
  comparison?: {
    label: string;
    value: number;
    type: 'previous' | 'target' | 'benchmark';
  };
  status?: 'good' | 'warning' | 'danger' | 'neutral';
  breakdown?: Array<{
    label: string;
    value: number;
    percentage: number;
  }>;
  onClick?: () => void;
}

const MetricCard: React.FC<MetricCardProps> = ({
  title,
  value,
  formatter = 'number',
  trend,
  comparison,
  status = 'neutral',
  breakdown,
  onClick
}) => {
  const getStatusColor = (status: string) => {
    const colors = {
      good: 'border-green-200 bg-green-50',
      warning: 'border-yellow-200 bg-yellow-50',
      danger: 'border-red-200 bg-red-50',
      neutral: 'border-gray-200 bg-white'
    };
    return colors[status] || colors.neutral;
  };
  
  const formatValue = (val: number): string => {
    switch (formatter) {
      case 'currency':
        return new Intl.NumberFormat('zh-CN', {
          style: 'currency',
          currency: 'CNY',
          minimumFractionDigits: 0,
          maximumFractionDigits: 0
        }).format(val);
      case 'percentage':
        return `${(val * 100).toFixed(2)}%`;
      default:
        return val.toLocaleString();
    }
  };
  
  return (
    <Card 
      className={`metric-card ${getStatusColor(status)} cursor-pointer hover:shadow-lg transition-shadow`}
      onClick={onClick}
      size="small"
    >
      <div className="space-y-3">
        {/* 标题行 */}
        <div className="flex items-center justify-between">
          <Text type="secondary" className="text-sm font-medium">
            {title}
          </Text>
          <StatusIndicator status={status} />
        </div>
        
        {/* 主要数值 */}
        <div className="space-y-1">
          <div className="text-2xl font-bold text-gray-900">
            {formatValue(value)}
          </div>
          
          {/* 趋势指示器 */}
          {trend && (
            <div className="flex items-center gap-1">
              <TrendIcon direction={trend.direction} />
              <Text className={`text-sm ${
                trend.direction === 'up' ? 'text-green-600' : 
                trend.direction === 'down' ? 'text-red-600' : 'text-gray-600'
              }`}>
                {formatValue(Math.abs(trend.value))} ({trend.period})
              </Text>
            </div>
          )}
          
          {/* 对比数据 */}
          {comparison && (
            <div className="text-sm text-gray-600">
              vs {comparison.label}: {formatValue(comparison.value)}
            </div>
          )}
        </div>
        
        {/* 分解数据 */}
        {breakdown && (
          <div className="space-y-1">
            {breakdown.map((item, index) => (
              <div key={index} className="flex justify-between text-xs">
                <span className="text-gray-600">{item.label}</span>
                <span className="font-medium">
                  {formatValue(item.value)} ({item.percentage.toFixed(1)}%)
                </span>
              </div>
            ))}
          </div>
        )}
      </div>
    </Card>
  );
};
```

### 4.1 参数调整交互

```tsx
// 参数调整的交互流程
const ParameterAdjustmentFlow = {
  // 1. 用户输入参数
  onUserInput: async (paramName: string, value: number) => {
    // 显示加载状态
    setLoadingState(paramName, true);
    
    // 实时验证
    const validation = await validateParameter(paramName, value);
    if (validation.hasErrors) {
      showValidationErrors(validation.errors);
      return;
    }
    
    // 显示预期影响
    const impact = await calculateImpactPreview(paramName, value);
    showImpactPreview(impact);
    
    // 防抖后执行计算
    debouncedCalculation(paramName, value);
  },
  
  // 2. 执行联动计算
  onCalculationComplete: (results: CalculationResults) => {
    // 更新所有受影响的组件
    updateAffectedComponents(results.affectedParameters);
    
    // 显示变化动画
    animateValueChanges(results.changes);
    
    // 更新图表
    updateCharts(results.chartData);
    
    // 显示计算摘要
    showCalculationSummary(results.summary);
  },
  
  // 3. 错误处理
  onCalculationError: (error: CalculationError) => {
    // 回滚到之前的值
    rollbackToPreviousValue();
    
    // 显示错误信息
    showErrorMessage(error.message);
    
    // 提供修复建议
    showFixSuggestions(error.suggestions);
  }
};
```
